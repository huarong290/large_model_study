# 2. 手动选择模型
# ================================================================

# 使用 Ollama
curl "http://localhost:12103/api/chat/select?model=ollama&message=你好"

# 使用 DeepSeek
curl "http://localhost:12103/api/chat/select?model=deepseek&message=你好"

# 使用 Gemini
curl "http://localhost:12103/api/chat/select?model=gemini&message=你好"

# 自动路由
curl "http://localhost:12103/api/chat/select?model=auto&message=你好"

# ================================================================
# 2. 自动路由
# ================================================================

# 代码场景 → DeepSeek
curl "http://localhost:12103/api/chat/route?message=写一段Java代码"

# 创意场景 → Gemini
curl "http://localhost:12103/api/chat/route?message=写一首关于春天的诗"

# 简单问答 → Ollama
curl "http://localhost:12103/api/chat/route?message=你好"
