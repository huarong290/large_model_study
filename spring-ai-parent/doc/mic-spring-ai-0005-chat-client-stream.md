# ================================================================
# 1. 流式响应测试
# ================================================================

# 基础流式响应
curl -N "http://localhost:12105/api/stream/chat?message=讲个故事"

# 带 System 消息
curl -N "http://localhost:12105/api/stream/with-system?message=你好&system=你是一个编程专家"

# 带超时控制
curl -N "http://localhost:12105/api/stream/chat-timeout?message=讲个故事&timeout=10"

# 带进度指示
curl -N "http://localhost:12105/api/stream/with-progress?message=写一篇关于人工智能的短文"

# 带统计信息
curl -N "http://localhost:12105/api/stream/with-stats?message=讲个故事"

# ================================================================
