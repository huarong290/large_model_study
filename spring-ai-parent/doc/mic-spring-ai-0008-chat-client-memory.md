# ================================================================
# 1. 带记忆的对话
# ================================================================

curl "http://localhost:12108/api/memory/chat?sessionId=user1&message=我叫张三"
curl "http://localhost:12108/api/memory/chat?sessionId=user1&message=我叫什么名字？"
curl "http://localhost:12108/api/memory/chat?sessionId=user1&message=我住在北京"
curl "http://localhost:12108/api/memory/chat?sessionId=user1&message=我住在哪里？"

# ================================================================
# 2. 不同 sessionId 独立
# ================================================================

curl "http://localhost:12108/api/memory/chat?sessionId=user1&message=我叫张三"
curl "http://localhost:12108/api/memory/chat?sessionId=user1&message=我叫什么名字？"
curl "http://localhost:12108/api/memory/chat?sessionId=user2&message=我叫李四"
curl "http://localhost:12108/api/memory/chat?sessionId=user2&message=我叫什么名字？"

# ================================================================
# 3. 无记忆的对话（对比）
# ================================================================

curl "http://localhost:12108/api/memory/no-memory?message=我叫张三"
curl "http://localhost:12108/api/memory/no-memory?message=我叫什么名字？"

# ================================================================
# 4. 批量对话
# ================================================================

curl -X POST "http://localhost:12108/api/memory/batch?sessionId=user1" \
-H "Content-Type: application/json" \
-d '["我叫张三", "我住在北京", "我的职业是程序员", "请总结我的信息"]'

# ================================================================
# 5. 获取历史记录
# ================================================================

curl "http://localhost:12108/api/memory/history?sessionId=user1"

# ================================================================
# 6. 清除记忆
# ================================================================

curl -X DELETE "http://localhost:12108/api/memory/clear?sessionId=user1"
curl -X DELETE "http://localhost:12108/api/memory/clear-all"

# ================================================================
# 7. 统计信息
# ================================================================

curl "http://localhost:12108/api/memory/stats"

# ================================================================
# 8. 对比演示
# ================================================================

curl "http://localhost:12108/api/memory/compare?sessionId=user1&name=张三&question=我叫什么名字？"
