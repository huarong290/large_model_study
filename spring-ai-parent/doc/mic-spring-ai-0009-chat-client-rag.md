# ================================================================
# 1. 构建知识库
# ================================================================

curl -X POST "http://localhost:12109/api/rag/build" \
-H "Content-Type: application/json" \
-d '{
"text": "Spring AI 是一个用于构建 AI 应用程序的框架。它提供了统一的 API 来访问各种 AI 模型，包括 OpenAI、Ollama、DeepSeek、Google Gemini。RAG（检索增强生成）是 Spring AI 的重要功能，支持向量数据库集成，包括 PGVector、Redis、Milvus 等。",
"title": "Spring AI 介绍",
"source": "用户输入"
}'

# ================================================================
# 2. RAG 问答
# ================================================================

curl -X POST "http://localhost:12109/api/rag/ask" \
-H "Content-Type: application/json" \
-d '{"question": "Spring AI 是什么？"}'

curl -X POST "http://localhost:12109/api/rag/ask" \
-H "Content-Type: application/json" \
-d '{"question": "Spring AI 支持哪些向量数据库？"}'
