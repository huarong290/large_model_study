# ================================================================
# 1. 安全对话（正常请求：带正确 API Key）
# ================================================================
curl -X POST "http://localhost:12115/api/security/chat" \
-H "Content-Type: application/json" \
-H "X-API-Key: admin-key-123" \
-d '{"message": "你好"}'

# ================================================================
# 2. 无安全控制对话（对比接口，不要求 API Key）
# ================================================================
curl -X POST "http://localhost:12115/api/security/chat/no-security" \
-H "Content-Type: application/json" \
-d '{"message": "你好"}'

# ================================================================
# 3. 拦截测试 A：未带 API Key（预期返回 401 Unauthorized）
# ================================================================
curl -X POST "http://localhost:12115/api/security/chat" \
-H "Content-Type: application/json" \
-d '{"message": "你好"}'

# ================================================================
# 4. 拦截测试 B：敏感词拦截（预期返回 400 Bad Request）
# ================================================================
curl -X POST "http://localhost:12115/api/security/chat" \
-H "Content-Type: application/json" \
-H "X-API-Key: admin-key-123" \
-d '{"message": "这是一个包含暴力的测试"}'

# ================================================================
# 5. 拦截测试 C：限流拦截（连续快速执行 11 次以上，预期返回 429 Too Many Requests）
# ================================================================
for i in {1..11}; do
curl -X POST "http://localhost:12115/api/security/chat" \
-H "Content-Type: application/json" \
-H "X-API-Key: admin-key-123" \
-d '{"message": "测试限流"}'
echo ""
done

# ================================================================
# 6. 监控与统计接口
# ================================================================
# 健康检查
curl "http://localhost:12115/api/security/health"

# 查看防护拦截统计（已拦截次数等）
curl "http://localhost:12115/api/security/stats"
