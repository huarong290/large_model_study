# ================================================================
# 1. 语音转文字（单文件 + 专有名词提示）
# ================================================================

curl -X POST "http://localhost:12118/api/audio/transcribe" \
-H "X-API-Key: admin-key-123" \
-F "file=@/path/to/audio.mp3" \
-F "language=zh" \
-F "prompt=专有名词：Spring AI, DeepSeek, Docker, RAG"

# ================================================================
# 2. 批量语音转文字
# ================================================================

curl -X POST "http://localhost:12118/api/audio/transcribe/batch" \
-H "X-API-Key: admin-key-123" \
-F "files=@/path/to/audio1.mp3" \
-F "files=@/path/to/audio2.mp3" \
-F "language=zh"

# ================================================================
# 3. 获取支持的格式
# ================================================================

curl -X GET "http://localhost:12118/api/audio/formats" \
-H "X-API-Key: admin-key-123"

# ================================================================
# 4. 健康检查（白名单路径，可免 Key 访问）
# ================================================================

curl -X GET "http://localhost:12118/api/audio/health"
